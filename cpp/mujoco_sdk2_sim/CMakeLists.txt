cmake_minimum_required(VERSION 3.16)
project(go2_mujoco_sdk2_sim LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(unitree_sdk2 REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenGL REQUIRED)

find_package(mujoco QUIET)
if(TARGET mujoco::mujoco)
  set(MJX_DEPLOY_MUJOCO_TARGET mujoco::mujoco)
else()
  find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h HINTS ${MUJOCO_ROOT} ENV MUJOCO_ROOT PATH_SUFFIXES include)
  find_library(MUJOCO_LIBRARY NAMES mujoco libmujoco.so.3.8.0 HINTS ${MUJOCO_ROOT} ENV MUJOCO_ROOT PATH_SUFFIXES lib bin "")
  if(NOT MUJOCO_LIBRARY)
    set(MJX_DEPLOY_MUJOCO_ROOTS ${MUJOCO_ROOT} $ENV{MUJOCO_ROOT})
    foreach(MJX_DEPLOY_ROOT ${MJX_DEPLOY_MUJOCO_ROOTS})
      if(MJX_DEPLOY_ROOT)
        file(GLOB MJX_DEPLOY_MUJOCO_CANDIDATES
          "${MJX_DEPLOY_ROOT}/libmujoco.so*"
          "${MJX_DEPLOY_ROOT}/lib/libmujoco.so*"
          "${MJX_DEPLOY_ROOT}/bin/libmujoco.so*")
        list(LENGTH MJX_DEPLOY_MUJOCO_CANDIDATES MJX_DEPLOY_MUJOCO_CANDIDATE_COUNT)
        if(MJX_DEPLOY_MUJOCO_CANDIDATE_COUNT GREATER 0)
          list(GET MJX_DEPLOY_MUJOCO_CANDIDATES 0 MUJOCO_LIBRARY)
          break()
        endif()
      endif()
    endforeach()
  endif()
  if(NOT MUJOCO_INCLUDE_DIR OR NOT MUJOCO_LIBRARY)
    message(FATAL_ERROR "未找到 MuJoCo。请安装 libmujoco，或设置 MUJOCO_ROOT 指向 MuJoCo 安装目录。")
  endif()
  get_filename_component(MJX_DEPLOY_MUJOCO_LIBRARY_DIR "${MUJOCO_LIBRARY}" DIRECTORY)
endif()

find_package(glfw3 QUIET)
if(TARGET glfw)
  set(MJX_DEPLOY_GLFW_TARGET glfw)
elseif(TARGET glfw3)
  set(MJX_DEPLOY_GLFW_TARGET glfw3)
else()
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(GLFW3 QUIET glfw3)
  endif()
  if(GLFW3_FOUND)
    set(MJX_DEPLOY_GLFW_INCLUDE_DIRS ${GLFW3_INCLUDE_DIRS})
    set(MJX_DEPLOY_GLFW_LIBRARIES ${GLFW3_LIBRARIES})
  else()
    find_path(GLFW_INCLUDE_DIR GLFW/glfw3.h)
    find_library(GLFW_LIBRARY NAMES glfw glfw3)
    if(NOT GLFW_INCLUDE_DIR OR NOT GLFW_LIBRARY)
      message(FATAL_ERROR "未找到 GLFW3。请安装 libglfw3-dev。")
    endif()
    set(MJX_DEPLOY_GLFW_INCLUDE_DIRS ${GLFW_INCLUDE_DIR})
    set(MJX_DEPLOY_GLFW_LIBRARIES ${GLFW_LIBRARY})
  endif()
endif()

add_executable(go2_mujoco_sdk2_sim src/go2_mujoco_sdk2_sim.cpp)
target_link_libraries(go2_mujoco_sdk2_sim PRIVATE unitree_sdk2 Threads::Threads OpenGL::GL ${CMAKE_DL_LIBS})

if(TARGET mujoco::mujoco)
  target_link_libraries(go2_mujoco_sdk2_sim PRIVATE mujoco::mujoco)
else()
  target_include_directories(go2_mujoco_sdk2_sim PRIVATE ${MUJOCO_INCLUDE_DIR})
  target_link_libraries(go2_mujoco_sdk2_sim PRIVATE ${MUJOCO_LIBRARY})
  set_target_properties(go2_mujoco_sdk2_sim PROPERTIES
    BUILD_RPATH "${MJX_DEPLOY_MUJOCO_LIBRARY_DIR}"
    INSTALL_RPATH "${MJX_DEPLOY_MUJOCO_LIBRARY_DIR}")
endif()

if(MJX_DEPLOY_GLFW_TARGET)
  target_link_libraries(go2_mujoco_sdk2_sim PRIVATE ${MJX_DEPLOY_GLFW_TARGET})
else()
  target_include_directories(go2_mujoco_sdk2_sim PRIVATE ${MJX_DEPLOY_GLFW_INCLUDE_DIRS})
  target_link_libraries(go2_mujoco_sdk2_sim PRIVATE ${MJX_DEPLOY_GLFW_LIBRARIES})
endif()
